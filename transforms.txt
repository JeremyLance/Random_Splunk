# Catches ANY sourcetype ending in -too_small or a trailing -<digits>
# (aide-2, aide-6, debug-21, instance-state-15, cron-too_small,
#  sudo-too_small, waagent-4, CommandExecution-3, etc. all match this)
[_rule:ruleset_generic_suffix_drop:filter:regex:generic01]
INGEST_EVAL = queue=if(match(sourcetype, "-(too_small|\d+)$"), "nullQueue", queue)
STOP_PROCESSING_IF = queue == "nullQueue"

# Catches sourcetypes with irregular / non-suffix naming that still need
# to be dropped outright (exact literal matches, alternated in one regex)
[_rule:ruleset_literal_name_drop:filter:regex:literal01]
INGEST_EVAL = queue=if(match(sourcetype, "^(too_small|aide\.log|allwin|boot|oneagent_latest_snapshot|node:sidecar:agent_manager:stdout|node:sidecar:cmp_orchestrator|node:sidecar:cmp_orchestrator:stdout|node:sidecar:ipc_broker:stdout|node:sidecar:postgres:stderr|secure_gateway_app_internal_log|diag_executor|dbx_server_connection_pool_metrics|dbx_server_health_metrics|_var_log_amazon_ssm_amazon-ssm-agent-too_small|_var_log_audit_audit-too_small|_var_log_messages-too_small|_opt_apigee_var_log_edge-message-processor_edge-message-processor-too_small|_Quadient_InspireScaler_logs_server-too_small|hadoop-yarn-resourcemanager-ip-10-157-116-74\.healthcareit\.net-too_small)$"), "nullQueue", queue)
STOP_PROCESSING_IF = queue == "nullQueue"

# Global metrics setting (unrelated to drop logic, kept as-is)
[_ruleset:global_settings]
metrics.disabled = false
