[mcds_soap_drop_empty_markers]
REGEX = ^\s*\*\s*$
DEST_KEY = queue
FORMAT = nullQueue

[mcds_soap_drop_stars]
REGEX = ^[\s*]+$
DEST_KEY = queue
FORMAT = nullQueue

LINE_BREAKER = ([\r\n]+)\*?(?=SOAP (?:REQUEST RECEIVED|RESPONSE))
TIME_PREFIX = TIME:\s+
TIME_FORMAT = %d/%b/%Y:%H:%M:%S
MAX_TIMESTAMP_LOOKAHEAD = 128
TRUNCATE = 0
SEDCMD-strip-stars = ^[\s*]+$
SEDCMD-strip-star-prefix = s/^\*+(SOAP (REQUEST RECEIVED|RESPONSE))/\1/mg
TRANSFORMS-drop_empty_markers = mcds_soap_drop_empty_markers
TRANSFORMS-drop_stars = mcds_soap_drop_stars
