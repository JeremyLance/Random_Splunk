## DEBUG ROUTING (OPTIONAL BUT VERY USEFUL)
############################################################
[debug_events]

DEST_KEY = _MetaData:Index
FORMAT = debug_linux_pipeline

############################################################
## GLOBAL SOURCETYPE NORMALIZATION
############################################################
[fix_bad_sourcetype]
REGEX = ([-\s]+[^-\s]+_?)(?:-\d+-too_small.*-20\d{2}.*)?$
DEST_KEY = MetaData:Sourcetype
FORMAT = sourcetype::$1

############################################################
## AUDIT LOGS
############################################################
[force_audit_sourcetype]
REGEX = /var/log/audit/audit\.log(\.\d+)?
DEST_KEY = MetaData:Sourcetype
FORMAT = sourcetype::linux:audit
WRITE_META = true

[force_audit_index]
REGEX = /var/log/audit/audit\.log(\.\d+)?
DEST_KEY = _MetaData:Index
FORMAT = sec_os_uhgid-000024

############################################################
## SYSLOG FAMILY
############################################################
[force_syslog_sourcetype]
REGEX = /var/log/(secure|messages|cron|syslog|kern\.log)(\.\d+)?
DEST_KEY = MetaData:Sourcetype
FORMAT = sourcetype::linux:syslog
WRITE_META = true

[force_syslog_index]
REGEX = /var/log/(secure|messages|cron|syslog|kern\.log)(\.\d+)?
DEST_KEY = _MetaData:Index
FORMAT = sec_os_uhgid-000024

############################################################
## NGINX
############################################################

[force_nginx_error_sourcetype]

[force_nginx_index]

############################################################

############################################################

[remove_too_small_suffix]

DEST_KEY = MetaData:Sourcetype

[linux_syslog_header]
REGEX = (?<syslog_time>\w{3}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+(?<host>\S+)\s+(?<process>[^\[]+)\[(?<pid>\d+)\]:

[linux_audit_header]
REGEX = type=(?<audit_type>\w+)\s+msg=audit\((?<audit_epoch>\d+)\.(?<audit_ms>\d+):(?<audit_id>\d+)\)

[linux_aide_fields]
REGEX = (?<aide_time>\d{2}-\d{2}-\d{4}\s+\d{2}:\d{2}:\d{2}\.\d{3}\s+[+-]\d{4})\s+(?<host>\w+)

[linux_access_combined]
REGEX = (?<client_ip>\S+)\s+\S+\s+\S+\s+\[(?<timestamp>[^\]]+)\]\s+"(?<method>\S+)\s+(?<uri>\S+)\s*(?<http_version>[^"]*)"\s+(?<status>\d+)\s+(?<bytes>\d+)
