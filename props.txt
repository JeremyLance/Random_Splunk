########################################################
# LINUX SYSLOG (default for most logs)
########################################################

[linux:syslog]
SHOULD_LINEMERGE = false
LINE_BREAKER = ([\r\n]+)(?=\w{3}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})
TRUNCATE = 200000
MAX_EVENTS = 10000
TIME_PREFIX = ^
TIME_FORMAT = %b %d %H:%M:%S
MAX_TIMESTAMP_LOOKAHEAD = 20
MAX_DIFF_SECS_AGO = 31536000
MAX_DIFF_SECS_HENCE = 31536000
KV_MODE = auto
REPORT-syslog_header = linux_syslog_header


########################################################
# LINUX AUDIT
########################################################

[linux:audit]

SHOULD_LINEMERGE = false

LINE_BREAKER = ([\r\n]+)(?=type=)

TRUNCATE = 200000
MAX_EVENTS = 10000

TIME_PREFIX = msg=audit\(
TIME_FORMAT = %s
MAX_TIMESTAMP_LOOKAHEAD = 20

KV_MODE = auto

REPORT-audit_header = linux_audit_header


########################################################
# LINUX AIDE
########################################################

[linux:aide]

SHOULD_LINEMERGE = false

LINE_BREAKER = ([\r\n]*)(?=\d{2}-\d{2}-\d{4}\s+\d{2}:\d{2}:\d{2}\.\d{3})

TRUNCATE = 200000
MAX_EVENTS = 10000

TIME_PREFIX = ^
TIME_FORMAT = %m-%d-%Y %H:%M:%S.%3N %z
MAX_TIMESTAMP_LOOKAHEAD = 30

KV_MODE = auto

REPORT-aide_fields = linux_aide_fields
