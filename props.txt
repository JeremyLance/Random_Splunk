########################################################
# LINUX SYSLOG (primary for most /var/log files)
########################################################
[source::...]
TRANSFORMS-01-clean = fix_bad_sourcetype

[source::/var/log/*]
TRANSFORMS-01-clean = fix_bad_sourcetype

[source::/var/log/cron]
TRANSFORMS-01-clean = fix_bad_sourcetype

[linux:syslog]
SHOULD_LINEMERGE = false
# Break on syslog timestamps only
LINE_BREAKER = ([\r\n]+)(?=\w{3}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})
TRUNCATE = 200000
MAX_EVENTS = 10000
TIME_PREFIX = ^
TIME_FORMAT = %b %d %H:%M:%S
MAX_TIMESTAMP_LOOKAHEAD = 20
MAX_DIFF_SECS_AGO = 31536000
MAX_DIFF_SECS_HENCE = 31536000
KV_MODE = auto
REPORT-syslog_header = linux_syslog_header
TRANSFORMS-01-clean = fix_bad_sourcetype

########################################################
# LINUX AUDIT (auditd logs)
########################################################

[linux_audit]
SHOULD_LINEMERGE = false
# Each audit event starts with "type="
LINE_BREAKER = ([\r\n]+)(?=type=)
TRUNCATE = 200000
MAX_EVENTS = 10000
# Extract epoch timestamp from msg=audit(...)
TIME_PREFIX = msg=audit\(
TIME_FORMAT = %s
MAX_TIMESTAMP_LOOKAHEAD = 20
KV_MODE = auto
REPORT-audit_header = linux_audit_header
TRANSFORMS-01-clean = fix_bad_sourcetype

########################################################
# LINUX AIDE (file integrity logs)
########################################################

[linux:aide]
SHOULD_LINEMERGE = false
# Break on timestamp (even if no newline exists)
LINE_BREAKER = ([\r\n]*)(?=\d{2}-\d{2}-\d{4}\s+\d{2}:\d{2}:\d{2}\.\d{3})
TRUNCATE = 200000
MAX_EVENTS = 10000
TIME_PREFIX = ^
TIME_FORMAT = %m-%d-%Y %H:%M:%S.%3N %z
MAX_TIMESTAMP_LOOKAHEAD = 30
MAX_DIFF_SECS_AGO = 31536000
MAX_DIFF_SECS_HENCE = 31536000
KV_MODE = auto
REPORT-aide_fields = linux_aide_fields
TRANSFORMS-01-clean = fix_bad_sourcetype

########################################################
# HTTP ACCESS LOGS (nginx/apache)
########################################################

[access_combined]
SHOULD_LINEMERGE = false
LINE_BREAKER = ([\r\n]+)
TRUNCATE = 200000
TIME_PREFIX = \[
TIME_FORMAT = %d/%b/%Y:%H:%M:%S %z
MAX_TIMESTAMP_LOOKAHEAD = 30
KV_MODE = none
REPORT-access = linux_access_combined
TRANSFORMS-01-clean = fix_bad_sourcetype
